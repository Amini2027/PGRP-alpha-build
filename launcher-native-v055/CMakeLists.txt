cmake_minimum_required(VERSION 3.22)
project(PGRP_NativeLauncher LANGUAGES CXX)
add_executable(PGRPLauncher WIN32 pgrp_launcher.cpp)
target_compile_features(PGRPLauncher PRIVATE cxx_std_17)
if(MSVC)
  set_property(TARGET PGRPLauncher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(PGRPLauncher PRIVATE /O2 /EHsc /W3 /permissive- /utf-8)
  target_link_options(PGRPLauncher PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO)
endif()
target_link_libraries(PGRPLauncher PRIVATE d2d1 dwrite ole32 shell32 comdlg32 ws2_32 user32)
